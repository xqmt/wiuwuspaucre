local function GradientText(text, fromColor, toColor)
	local result = ""
	for i = 1, #text do
		local t = (i - 1) / math.max(#text - 1, 1)
		local r = math.floor((fromColor.R + (toColor.R - fromColor.R) * t) * 255)
		local g = math.floor((fromColor.G + (toColor.G - fromColor.G) * t) * 255)
		local b = math.floor((fromColor.B + (toColor.B - fromColor.B) * t) * 255)
		result = result .. string.format('<font color="rgb(%d,%d,%d)">%s</font>', r, g, b, text:sub(i, i))
	end
	return result
end
